cmake_minimum_required(VERSION 3.16)
project(echo_ggwave LANGUAGES CXX)
add_library(echo_ggwave SHARED Source/echo_ggwave.cpp ../ThirdParty/ggwave/src/ggwave.cpp)
target_include_directories(echo_ggwave PRIVATE ../ThirdParty/ggwave/include ../ThirdParty/ggwave/src)
target_compile_features(echo_ggwave PRIVATE cxx_std_11)
target_compile_definitions(echo_ggwave PRIVATE GGWAVE_DISABLE_LOG)
if(MSVC)
  target_compile_options(echo_ggwave PRIVATE /O2)
else()
  target_compile_options(echo_ggwave PRIVATE -O2 -fvisibility=hidden)
endif()
